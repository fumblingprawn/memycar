'use client';

import { notFound } from 'next/navigation';
import { useEffect, useState } from 'react';
import Image from 'next/image';
import Link from 'next/link';
import { supabase } from '@/lib/supabase/client';
import { Listing } from '@/types/listing';

const ListingDetailPage: React.FC<{ params: { id: string } }> = ({ params }) => {
  const [listing, setListing] = useState<Listing | null>(null);
  const [loading, setLoading] = useState<boolean>(true);
  const [error, setError] = useState<string | null>(null);
  const [currentImageIndex, setCurrentImageIndex] = useState<number>(0);

  // Fetch listing by ID
  useEffect(() => {
    const fetchListing = async () => {
      setLoading(true);
      setError(null);

      try {
        const { data, error: supabaseError } = await supabase
          .from('listings')
          .select('*')
          .eq('id', params.id)
          .single();

        if (supabaseError) throw supabaseError;
        if (!data) {
          notFound();
          return;
        }

        setListing(data as Listing);
        // Reset to first image when listing changes
        setCurrentImageIndex(0);
      } catch (err) {
        console.error('Error fetching listing:', err);
        setError('Failed to load listing. Please try again later.');
        notFound();
      } finally {
        setLoading(false);
      }
    };

    fetchListing();
  }, [params.id]);

  if (loading) {
    return (
      <div className="min-h-screen bg-gray-50 flex items-center justify-center py-12">
        <div className="animate-spin rounded-full h-10 w-10 border-b-2 border-blue-500"></div>
      </div>
    );
  }

  if (error || !listing) {
    return (
      <div className="min-h-screen bg-gray-50 flex items-center justify-center py-12">
        <p className="text-gray-500">Listing not found</p>
      </div>
    );
  }

  // Get all images in display order: 5 mandatory + extras
  const getAllImages = (): string[] => {
    const mandatorySlots: PhotoSlotKey[] = [
      'front_three_quarter',
      'rear_three_quarter',
      'side_profile',
      'interior_dash',
      'odometer'
    ];

    const images: string[] = [];

    // Add mandatory slots in order
    mandatorySlots.forEach(slot => {
      if (listing.photos[slot]) {
        images.push(listing.photos[slot]);
      }
    });

    // Add extra photos
    if (listing.photos.extra_photos) {
      images.push(...listing.photos.extra_photos);
    }

    return images;
  };

  const allImages = getAllImages();
  const hasImages = allImages.length > 0;
  const currentImage = hasImages ? allImages[currentImageIndex] : null;

  // Format phone number for WhatsApp (remove non-digits)
  const cleanWhatsApp = listing.seller_whatsapp.replace(/\D/g, '');

  // Pre-filled WhatsApp message
  const whatsappMessage = encodeURIComponent(
    `Hi ${listing.seller_name}, I saw your ${listing.year} ${listing.make} ${listing.model} on memycar.com (${window.location.href}). Is it still available?`
  );

  return (
    <div className="min-h-screen bg-gray-50">
      {/* Mobile Sticky Action Bar */}
      <div className="fixed bottom-0 left-0 right-0 bg-white border-t shadow-md hidden sm:block pb-4 px-4">
        <div className="max-w-4xl mx-auto flex justify-between items-center">
          <a
            href={`https://wa.me/${cleanWhatsApp}?text=${whatsappMessage}`}
            className="flex-1 bg-green-600 hover:bg-green-700 text-white font-medium py-3 px-4 rounded-md text-center text-sm"
          >
            WhatsApp Seller
          </a>
          <a
            href={`tel:${listing.seller_phone.replace(/\s/g, '')}`}
            className="flex-1 ml-4 border border-gray-300 hover:bg-gray-50 text-gray-700 font-medium py-3 px-4 rounded-md text-center text-sm"
          >
            Call Seller
          </a>
        </div>
      </div>

      {/* Desktop Content */}
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 pt-16 pb-12">
        {/* Sidebar on Desktop */}
        <div className="hidden md:flex md:w-1/4">
          <div className="sticky top-16">
            {/* Lead Action Bar (Desktop Sidebar) */}
            <div className="mb-6">
              <a
                href={`https://wa.me/${cleanWhatsApp}?text=${whatsappMessage}`}
                className="w-full bg-green-600 hover:bg-green-700 text-white font-medium py-3 px-4 rounded-md text-center text-sm block"
              >
                WhatsApp Seller
              </a>
              <a
                href={`tel:${listing.seller_phone.replace(/\s/g, '')}`}
                className="w-full mt-3 border border-gray-300 hover:bg-gray-50 text-gray-700 font-medium py-3 px-4 rounded-md text-center text-sm block"
              >
                Call Seller
              </a>
            </div>

            {/* Seller Info */}
            <div className="bg-white rounded-lg shadow-md p-4">
              <h3 className="text-lg font-semibold text-gray-900 mb-3">
                Seller Information
              </h3>
              <div className="space-y-3">
                <div className="flex items-center">
                  <div className="flex-shrink-0 h-10 w-10 bg-blue-100 rounded-full flex items-center justify-center">
                    <svg className="h-5 w-5 text-blue-600" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
                      <path d="M20 21v-2a4 4 0 00-4-4H8a4 4 0 00-4 4v2"></path>
                      <circle cx="12" cy="7" r="4" />
                    </svg>
                  </div>
                  <div className="ml-3">
                    <p className="font-medium text-gray-900">{listing.seller_name}</p>
                    <p className="text-sm text-gray-500">
                      {listing.seller_phone} |
                      {listing.seller_whatsapp ? `WhatsApp: ${listing.seller_whatsapp}` : 'WhatsApp: Not provided'}
                    </p>
                  </div>
                </div>

                {/* Location Badge */}
                <div className="mt-3">
                  <span className="inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium bg-emerald-100 text-emerald-800">
                    {listing.emirate}
                  </span>
                </div>
              </div>
            </div>
          </div>
        </div>

        {/* Main Content */}
        <div className="md:ml-10 md:flex-1">
          {/* Photo Showcase */}
          <div className="mb-8">
            {/* Main Image Viewer */}
            <div className="relative w-full h-0 pt-[56.25%] bg-gray-200"> /* 16:9 ratio */
              {hasImages && currentImage ? (
                <Image
                  src={currentImage}
                  alt={`${listing.year} ${listing.make} ${listing.model} - Image ${currentImageIndex + 1} of ${allImages.length}`}
                  className="absolute inset-0 w-full h-full object-cover"
                  priority
                  onError={(e) => {
                    (e.currentTarget as HTMLImageElement).src = '/placeholder-car.jpg';
                    (e.currentTarget as HTMLImageElement).onerror = null;
                  }}
                />
              ) : (
                <div className="absolute inset-0 flex items-center justify-center text-gray-500">
                  No images available
                </div>
              )}

              {/* Navigation Arrows */}
              {hasImages && allImages.length > 1 && (
                <>
                  <button
                    onClick={() => setCurrentImageIndex((prev) => (prev - 1 + allImages.length) % allImages.length)}
                    className="absolute left-3 top-1/2 -translate-y-1/2 bg-white/70 hover:bg-white/90 rounded-full p-2 transition-colors z-10"
                  >
                    <svg className="h-5 w-5 text-gray-600" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
                      <path d="M15 18l-6-6 6-6" />
                    </svg>
                  </button>
                  <button
                    onClick={() => setCurrentImageIndex((prev) => (prev + 1) % allImages.length)}
                    className="absolute right-3 top-1/2 -translate-y-1/2 bg-white/70 hover:bg-white/90 rounded-full p-2 transition-colors z-10"
                  >
                    <svg className="h-5 w-5 text-gray-600" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
                      <path d="M9 6l6 6-6 6" />
                    </svg>
                  </button>
                </>
              )}

              {/* Thumbnail Slider */}
              {hasImages && allImages.length > 1 && (
                <div className="mt-3 flex justify-center space-x-2">
                  {allImages.map((img, index) => (
                    <button
                      key={index}
                      onClick={() => setCurrentImageIndex(index)}
                      className={`w-8 h-8 rounded border-2 ${
                        currentImageIndex === index
                          ? 'border-blue-500 bg-blue-50'
                          : 'border-gray-200 hover:border-gray-300'
                      }`}
                    >
                      <Image
                        src={img}
                        alt={`Thumbnail ${index + 1}`}
                        className="w-full h-full object-cover"
                        priority
                      />
                    </button>
                  ))}
                </div>
              )}
            </div>
          </div>

          {/* Clean Header & Spec Matrix */}
          <div className="mb-8">
            {/* Header */}
            <div className="mb-4">
              <h1 className="text-2xl font-bold text-gray-900">
                {listing.year} {listing.make} {listing.model}
                {listing.trim && ` ${listing.trim}`}
              </h1>
              <div className="flex items-baseline gap-4 mt-2">
                <span className="text-2xl font-bold text-emerald-600">
                  AED {listing.price_aed.toLocaleString()}
                </span>
                <div className="text-sm text-gray-500 flex items-center">
                  <time dateTime={listing.created_at}>
                    {new Date(listing.created_at).toLocaleDateString('en-GB', {
                      day: 'numeric',
                      month: 'short',
                      year: 'numeric'
                    })}
                  </time>
                  <span className="mx-2">•</span>
                  <span>{listing.emirate}</span>
                </div>
              </div>
            </div>

            {/* Spec Matrix - 3x2 Grid */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              {/* Mileage */}
              <div className="bg-white rounded-lg shadow-sm p-4 flex items-center space-x-3">
                <div className="flex-shrink-0 h-10 w-10 bg-blue-50 rounded-full flex items-center justify-center">
                  <svg className="h-5 w-5 text-blue-600" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
                    <path d="M12 2c-1.1 0-2 .9-2 2h1c0 .4.2.7.5.9l1 1c.5.5 1 .8 1.7.8h7c1.1 0 2-.9 2-2V5c0-1.1-.9-2-2-2h-1c-.4 0-.7.2-.9.5l-1-1c-.5-.5-.8-.1-.8-.6V4c0-.6.2-1.1.5-1.4l1-1c.2-.2.5-.3.8-.3h7c1.1 0 2 .9 2 2v1c0 .1-.1.1-.1.1z"/>
                  </svg>
                </div>
                <div>
                  <p className="text-sm font-medium text-gray-500">Mileage</p>
                  <p className="text-lg font-semibold text-gray-900">
                    {listing.mileage_km.toLocaleString()} km
                  </p>
                </div>
              </div>

              {/* Regional Specs */}
              <div className="bg-white rounded-lg shadow-sm p-4 flex items-center space-x-3">
                <div className="flex-shrink-0 h-10 w-10 bg-blue-50 rounded-full flex items-center justify-center">
                  <svg className="h-5 w-5 text-blue-600" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
                    <path d="M19 3H5c-1.1 0-2 .9-2 2v14c0 1.1.9 2 2 2h14c1.1 0 2-.9 2-2V5c0-1.1-.9-2-2-2z"/>
                    <path d="M8 9h8v10H8V9z"/>
                  </svg>
                </div>
                <div>
                  <p className="text-sm font-medium text-gray-500">Regional Specs</p>
                  <p className="text-lg font-semibold text-gray-900 capitalize">
                    {listing.specs === 'GCC' ? 'GCC' : 'Import'}
                  </p>
                </div>
              </div>

              {/* Service History */}
              <div className="bg-white rounded-lg shadow-sm p-4 flex items-center space-x-3">
                <div className="flex-shrink-0 h-10 w-10 bg-blue-50 rounded-full flex items-center justify-center">
                  <svg className="h-5 w-5 text-blue-600" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
                    <path d="M12 2C6.48 2 2 6.48 2 12s4.48 10 10 10 10-4.48 10-10S17.52 2 12 2zm0 18c-4.41 0-8-3.59-8-8s3.59-8 8-8 8 3.59 8-8-3.59 8-8-8z"/>
                  </svg>
                </div>
                <div>
                  <p className="text-sm font-medium text-gray-500">Service History</p>
                  <p className="text-lg font-semibold text-gray-900 capitalize">
                    {listing.service_history.replace(/\//g, ' ')}
                  </p>
                </div>
              </div>

              {/* Paint Condition */}
              <div className="bg-white rounded-lg shadow-sm p-4 flex items-center space-x-3">
                <div className="flex-shrink-0 h-10 w-10 bg-blue-50 rounded-full flex items-center justify-center">
                  <svg className="h-5 w-5 text-blue-600" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
                    <path d="M19 3H5c-1.1 0-2 .9-2 2v14c0 1.1.9 2 2 2h14c1.1 0 2-.9 2-2V5c0-1.1-.9-2-2-2z"/>
                    <path d="M12 11.2l-5 1.7v-.7l4.2-2.5L12 5l1.9 2.4v.7L12 11.2z"/>
                  </svg>
                </div>
                <div>
                  <p className="text-sm font-medium text-gray-500">Paint Condition</p>
                  <p className="text-lg font-semibold text-gray-900 capitalize">
                    {listing.paint_condition.replace(/\-/g, ' ')}
                  </p>
                </div>
              </div>

              {/* Warranty Status */}
              <div className="bg-white rounded-lg shadow-sm p-4 flex items-center space-x-3">
                <div className="flex-shrink-0 h-10 w-10 bg-blue-50 rounded-full flex items-center justify-center">
                  <svg className="h-5 w-5 text-blue-600" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
                    <path d="M12 2C6.48 2 2 6.48 2 12s4.48 10 10 10 10-4.48 10-10S17.52 2 12 2zm0 18c-4.41 0-8-3.59-8-8s3.59-8 8-8 8 3.59 8-8-3.59 8-8-8z"/>
                    <path d="M12 16l-4-4h3l-2-2 4 4h2l2-4h-3z"/>
                  </svg>
                </div>
                <div>
                  <p className="text-sm font-medium text-gray-500">Warranty Status</p>
                  <p className="text-lg font-semibold text-gray-900 capitalize">
                    {listing.warranty.replace(/\//g, ' ').replace(/_/g, ' ')}
                  </p>
                </div>
              </div>

              {/* Number of Keys */}
              <div className="bg-white rounded-lg shadow-sm p-4 flex items-center space-x-3">
                <div className="flex-shrink-0 h-10 w-10 bg-blue-50 rounded-full flex items-center justify-center">
                  <svg className="h-5 w-5 text-blue-600" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
                    <path d="M19 3H5c-1.1 0-2 .9-2 2v14c0 1.1.9 2 2 2h14c1.1 0 2-.9 2-2V5c0-1.1-.9-2-2-2z"/>
                    <path d="M9 17V7h6v10H9zm4-10v2h4V7h-4z"/>
                  </svg>
                </div>
                <div>
                  <p className="text-sm font-medium text-gray-500">Number of Keys</p>
                  <p className="text-lg font-semibold text-gray-900">
                    {listing.keys_count}
                  </p>
                </div>
              </div>
            </div>
          </div>

          {/* Structured Overview / Description */}
          <div className="mb-8">
            <h2 className="text-xl font-bold text-gray-900 mb-4">
              Vehicle Overview
            </h2>
            <p className="text-gray-700 leading-relaxed">
              {listing.description}
            </p>
          </div>
